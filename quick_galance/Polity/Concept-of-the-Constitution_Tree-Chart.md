|   +-- rigid <-> flexible
|   +-- federal <-> unitary
|   +-- parliamentary <-> presidential
|   +-- republican <-> monarchical
|   `-- constitutional supremacy <-> classic parliamentary sovereignty

|                   -> accountability -> independent remedy

+-- 6. INDIAN CONSTITUTIONAL IDENTITY
