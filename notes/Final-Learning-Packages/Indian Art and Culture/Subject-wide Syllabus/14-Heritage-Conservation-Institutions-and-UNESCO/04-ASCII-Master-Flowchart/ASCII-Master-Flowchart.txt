    +-- NAME THE OBJECT FIRST
    |      monument / site / landscape -> TANGIBLE
    |      ritual / festival / skill   -> INTANGIBLE
    |      manuscript / archive / file -> DOCUMENTARY
    |      GI = domestic IP, not UNESCO
    |
    +-- NAME THE METHOD NEXT
           conservation -> stabilise, retain fabric
